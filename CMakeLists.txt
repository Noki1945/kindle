cmake_minimum_required(VERSION 3.16)
project(KindleLiteOS VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kindle_core src/library/catalog.cpp src/reader/text_reader.cpp src/core/state.cpp src/eink/refresh.cpp)
add_library(kindle_web src/browser/web.cpp)
target_include_directories(kindle_web PUBLIC include)
if(MSVC)
  target_compile_options(kindle_web PRIVATE /utf-8)
endif()
add_library(book_dependencies third_party/tinyxml2/tinyxml2.cpp third_party/miniz/miniz.c third_party/miniz/miniz_tdef.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(book_dependencies PUBLIC third_party/tinyxml2 third_party/miniz)
target_compile_definitions(book_dependencies PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(kindle_documents src/reader/document.cpp)
target_include_directories(kindle_documents PUBLIC include)
target_link_libraries(kindle_documents PUBLIC book_dependencies)
target_include_directories(kindle_core PUBLIC include)
if(MSVC)
  target_compile_options(kindle_core PUBLIC /W4 /utf-8)
else()
  target_compile_options(kindle_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(kindle_cli src/core/main.cpp)
target_link_libraries(kindle_cli PRIVATE kindle_core)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(kindle_ui src/ui/kindle.cpp src/browser/linux.cpp)
  target_link_libraries(kindle_ui PRIVATE kindle_core)
  target_link_libraries(kindle_ui PRIVATE kindle_documents)
  target_link_libraries(kindle_ui PRIVATE kindle_web)
endif()
if(WIN32)
  add_executable(kindle_sim WIN32 src/ui/windows.cpp src/browser/winhttp.cpp)
  target_compile_definitions(kindle_sim PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(kindle_sim PRIVATE kindle_core user32 gdi32 comdlg32 winhttp)
  target_link_libraries(kindle_sim PRIVATE kindle_documents)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE kindle_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(document_tests tests/document_tests.cpp)
  target_link_libraries(document_tests PRIVATE kindle_documents kindle_core)
  if(MSVC)
    target_compile_options(kindle_documents PRIVATE /utf-8)
    target_compile_options(document_tests PRIVATE /utf-8)
  endif()
  add_test(NAME documents COMMAND document_tests)
  add_executable(web_tests tests/web_tests.cpp)
  target_link_libraries(web_tests PRIVATE kindle_web)
  if(MSVC)
    target_compile_options(web_tests PRIVATE /utf-8)
  endif()
  add_test(NAME web COMMAND web_tests)
  if(WIN32)
    add_executable(browser_tests tests/browser_tests.cpp src/browser/winhttp.cpp)
    target_compile_definitions(browser_tests PRIVATE NOMINMAX)
    target_link_libraries(browser_tests PRIVATE kindle_core winhttp)
    add_test(NAME browser_parser COMMAND browser_tests)
  endif()
endif()
